-- Section C — Numeric Filters

-- 10. Display stations whose capacity is greater than 100 .
SELECT *FROM c 
WHERE c.capacity_kw>100
-- 11. Display stations whose revenue is greater than 15000 .
SELECT *FROM c 
WHERE c.revenue>15000
-- 12. Display stations having more than 40 daily sessions.
SELECT *FROM c 
WHERE c.daily_sessions>40
-- 13. Display stations whose uptime is below 96 .
SELECT *FROM c 
WHERE c.uptime<96
-- 14. Display stations whose capacity is between 50 and 120 .
SELECT *FROM c 
WHERE c.capacity_kw>50 and c.capacity_kw<120
--
-- 15. Display stations whose revenue is below 10000 .
SELECT *FROM c 
WHERE c.revenue<10000
